cmake_minimum_required(VERSION 3.20)
project(CKInstallerTheme LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "NSIS Unicode uses 32-bit plugins; configure with -A Win32")
endif()
add_library(CKTheme SHARED theme.cpp theme.def)
target_compile_features(CKTheme PRIVATE cxx_std_17)
target_compile_definitions(CKTheme PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
target_link_libraries(CKTheme PRIVATE comctl32 uxtheme)
set_property(TARGET CKTheme PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(CKTheme PRIVATE /utf-8 /W4)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(theme-test theme-test.cpp)
  target_compile_definitions(theme-test PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0601 _WIN32_WINNT=0x0601)
  target_link_libraries(theme-test PRIVATE CKTheme comctl32)
  set_property(TARGET theme-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MSVC)
    target_compile_options(theme-test PRIVATE /utf-8)
  endif()
  add_test(NAME installer-theme COMMAND theme-test)
endif()
