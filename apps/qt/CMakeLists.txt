cmake_minimum_required(VERSION 3.20)
project(CKLauncher VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(CK_LEGACY_WINDOWS "Build the Qt 5 Windows 7/8.1 frontend" OFF)
if(CK_LEGACY_WINDOWS)
  find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets Network Test)
  set(QT Qt5)
else()
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Test)
  set(QT Qt6)
endif()
add_executable(ck-launcher-qt WIN32 src/main.cpp src/backend.cpp src/window.cpp src/library.cpp src/catalog.cpp src/accounts.cpp src/settings.cpp resources.qrc)
target_link_libraries(ck-launcher-qt PRIVATE ${QT}::Widgets ${QT}::Network)
target_compile_definitions(ck-launcher-qt PRIVATE CK_VERSION="0.2.0-beta.1" QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
  target_compile_options(ck-launcher-qt PRIVATE /utf-8 /W4)
else()
  target_compile_options(ck-launcher-qt PRIVATE -Wall -Wextra)
endif()
if(CK_LEGACY_WINDOWS)
  target_compile_definitions(ck-launcher-qt PRIVATE CK_LEGACY_WINDOWS WINVER=0x0601 _WIN32_WINNT=0x0601)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol-test tests/protocol.cpp src/backend.cpp)
  target_include_directories(protocol-test PRIVATE src)
  target_link_libraries(protocol-test PRIVATE ${QT}::Test ${QT}::Network ${QT}::Widgets)
  add_test(NAME protocol COMMAND protocol-test -platform offscreen)
endif()
install(TARGETS ck-launcher-qt RUNTIME DESTINATION .)
