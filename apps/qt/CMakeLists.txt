cmake_minimum_required(VERSION 3.20)
project(CKLauncher VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
option(CK_LEGACY_WINDOWS "Build the Qt 5 Windows 7/8.1 frontend" OFF)
if(CK_LEGACY_WINDOWS)
  find_package(Qt5 5.15 REQUIRED COMPONENTS Widgets Network Test)
  set(QT Qt5)
else()
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Test)
  set(QT Qt6)
endif()
add_library(ck-ui STATIC src/backend.cpp src/window.cpp src/library.cpp src/catalog.cpp src/accounts.cpp src/settings.cpp src/details.cpp src/ui.cpp src/skinview.cpp src/projectview.cpp src/emote.cpp)
target_include_directories(ck-ui PUBLIC src)
target_link_libraries(ck-ui PUBLIC ${QT}::Widgets ${QT}::Network)
if(WIN32)
  target_link_libraries(ck-ui PUBLIC winmm)
endif()
target_compile_definitions(ck-ui PUBLIC CK_VERSION="1.0.0" QT_NO_CAST_FROM_ASCII QT_NO_CAST_TO_ASCII)
if(MSVC)
  target_compile_options(ck-ui PUBLIC /utf-8 /W4)
else()
  target_compile_options(ck-ui PRIVATE -Wall -Wextra)
endif()
if(CK_LEGACY_WINDOWS)
  target_compile_definitions(ck-ui PUBLIC CK_LEGACY_WINDOWS WINVER=0x0601 _WIN32_WINNT=0x0601)
endif()
add_executable(ck-launcher-qt WIN32 src/main.cpp resources.qrc)
if(WIN32)
  configure_file("../../app/src-tauri/icons/icon.ico" launcher.ico COPYONLY)
  configure_file(launcher.rc.in launcher.rc @ONLY)
  target_sources(ck-launcher-qt PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/launcher.rc")
endif()
target_link_libraries(ck-launcher-qt PRIVATE ck-ui)
include(CTest)
if(BUILD_TESTING)
  add_executable(protocol-test tests/protocol.cpp src/backend.cpp)
  target_include_directories(protocol-test PRIVATE src)
  target_link_libraries(protocol-test PRIVATE ${QT}::Test ${QT}::Network ${QT}::Widgets)
  add_test(NAME protocol COMMAND protocol-test -platform offscreen)
  add_executable(ui-fixture tests/fixture.cpp)
  target_link_libraries(ui-fixture PRIVATE ${QT}::Gui)
  add_executable(ui-test tests/ui-test.cpp resources.qrc)
  target_link_libraries(ui-test PRIVATE ck-ui ${QT}::Test)
  add_dependencies(ui-test ui-fixture)
  add_executable(ui-preview WIN32 tests/preview.cpp resources.qrc)
  target_link_libraries(ui-preview PRIVATE ck-ui)
  add_dependencies(ui-preview ui-fixture)
  add_test(NAME interface COMMAND ui-test -platform offscreen)
  set_tests_properties(interface PROPERTIES TIMEOUT 90)
  add_executable(projectview-test tests/projectview-test.cpp resources.qrc)
  target_link_libraries(projectview-test PRIVATE ck-ui ${QT}::Test)
  add_dependencies(projectview-test ui-fixture)
  add_test(NAME project-versions COMMAND projectview-test -platform offscreen)
  set_tests_properties(project-versions PROPERTIES TIMEOUT 90)
  add_executable(emote-test tests/emote-test.cpp src/emote.cpp src/skinview.cpp resources.qrc)
  target_include_directories(emote-test PRIVATE src)
  target_link_libraries(emote-test PRIVATE ${QT}::Widgets ${QT}::Test)
  if(MSVC)
    target_compile_options(emote-test PRIVATE /utf-8)
  endif()
  add_test(NAME emotes COMMAND emote-test -platform offscreen)
  set_tests_properties(emotes PROPERTIES TIMEOUT 180)
endif()
install(TARGETS ck-launcher-qt RUNTIME DESTINATION .)
